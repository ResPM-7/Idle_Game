// Google Sheet에서 확장 프로그램 > Apps Script를 열고 이 코드를 붙여넣습니다.
// SPREADSHEET_ID와 ACCESS_TOKEN을 입력한 뒤 웹 앱으로 배포하세요.

const SPREADSHEET_ID = '여기에_구글_시트_ID_입력';
const ACCESS_TOKEN = 'Unity와_동일한_긴_인증_토큰_입력';
const ALLOWED_SHEETS = ['UnitData', 'EnemyData'];

/**
 * Unity 에디터가 POST로 보낸 아군/적군 데이터를 Google Sheet에 기록합니다.
 * 값만 지우고 다시 쓰므로 기존 셀 서식과 데이터 유효성 검사는 유지됩니다.
 */
function doPost(e) {
  const lock = LockService.getScriptLock();

  try {
    lock.waitLock(30000);

    const payload = JSON.parse(e.postData.contents);
    if (!payload.token || payload.token !== ACCESS_TOKEN) {
      throw new Error('인증 토큰이 올바르지 않습니다.');
    }

    if (!Array.isArray(payload.sheets) || payload.sheets.length === 0) {
      throw new Error('업로드할 시트 데이터가 없습니다.');
    }

    const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID);
    payload.sheets.forEach(sheetData => writeSheet(spreadsheet, sheetData));

    return jsonResponse({
      success: true,
      message: 'UnitData와 EnemyData 업로드가 완료되었습니다.'
    });
  } catch (error) {
    return jsonResponse({ success: false, message: String(error.message || error) });
  } finally {
    lock.releaseLock();
  }
}

function writeSheet(spreadsheet, sheetData) {
  if (!ALLOWED_SHEETS.includes(sheetData.sheetName)) {
    throw new Error('허용되지 않은 시트입니다: ' + sheetData.sheetName);
  }

  const sheet = spreadsheet.getSheetByName(sheetData.sheetName);
  if (!sheet) {
    throw new Error('시트를 찾지 못했습니다: ' + sheetData.sheetName);
  }

  const width = sheetData.headers.length;
  if (sheetData.types.length !== width) {
    throw new Error(sheetData.sheetName + ': 변수명과 자료형 개수가 다릅니다.');
  }

  const rows = [sheetData.headers, sheetData.types];
  sheetData.rows.forEach(row => {
    if (!row.values || row.values.length !== width) {
      throw new Error(sheetData.sheetName + ': 데이터 열 개수가 올바르지 않습니다.');
    }
    rows.push(convertTypes(row.values, sheetData.types));
  });

  // 이전 데이터가 더 길었을 수 있으므로 기존 범위까지 포함합니다.
  const clearRows = Math.max(sheet.getLastRow(), rows.length, 1);
  const clearColumns = Math.max(sheet.getLastColumn(), width, 1);
  const targetRange = sheet.getRange(1, 1, clearRows, clearColumns);
  const writeRange = sheet.getRange(1, 1, rows.length, width);

  // 업로드 도중 오류가 나도 원래 값을 되돌릴 수 있도록 먼저 백업합니다.
  const oldValues = targetRange.getValues();
  const oldValidations = targetRange.getDataValidations();

  // 엄격한 드롭다운은 Apps Script가 다중 값을 한 번에 기록할 때 먼저 거부할 수 있습니다.
  // 쓰는 동안만 검사를 제거하고, 값 입력이 끝나면 원래 규칙을 그대로 복원합니다.
  targetRange.clearDataValidations();
  try {
    targetRange.clearContent();
    writeRange.setValues(rows);
    targetRange.setDataValidations(oldValidations);
  } catch (error) {
    // 실패하면 빈 시트로 남지 않도록 업로드 직전 값과 규칙을 복구합니다.
    targetRange.clearContent();
    targetRange.setValues(oldValues);
    targetRange.setDataValidations(oldValidations);
    throw error;
  }
}

/**
 * Unity JSON에서는 모든 셀 값을 문자열로 보냅니다.
 * 2행의 자료형을 기준으로 숫자와 bool을 실제 Google Sheet 값으로 변환합니다.
 */
function convertTypes(values, types) {
  return values.map((value, index) => {
    const type = String(types[index] || '').replace(/\s/g, '').toLowerCase();

    // 배열과 UnitDataSO 참조 목록은 한 셀의 문자열 형태를 유지합니다.
    if (type.endsWith('[]') || type.startsWith('unitdataso')) {
      return value;
    }

    if (value === '') {
      return '';
    }

    if (type === 'int' || type === 'float' || type === 'long' || type === 'double') {
      const number = Number(value);
      if (!Number.isFinite(number)) {
        throw new Error('숫자로 변환할 수 없는 값입니다: ' + value);
      }
      return number;
    }

    if (type === 'bool') {
      return String(value).toLowerCase() === 'true' || value === '1';
    }

    return value;
  });
}

function jsonResponse(value) {
  return ContentService
    .createTextOutput(JSON.stringify(value))
    .setMimeType(ContentService.MimeType.JSON);
}
